#include <iostream>
#include <vector>
using namespace std;

#define V 4   // Number of vertices

// Function to check if it's safe to assign color c to vertex v
bool isSafe(int v, vector<vector<int>>& graph, vector<int>& color, int c) {
    for (int i = 0; i < V; i++) {
        if (graph[v][i] == 1 && color[i] == c)
            return false;
    }
    return true;
}

// Backtracking function
bool graphColoringUtil(vector<vector<int>>& graph, int m,
                       vector<int>& color, int v) {
    
    // If all vertices are assigned a color
    if (v == V)
        return true;

    // Try different colors
    for (int c = 1; c <= m; c++) {
        if (isSafe(v, graph, color, c)) {
            color[v] = c;

            if (graphColoringUtil(graph, m, color, v + 1))
                return true;

            // Backtrack
            color[v] = 0;
        }
    }

    return false;
}

void graphColoring(vector<vector<int>>& graph, int m) {
    vector<int> color(V, 0);

    if (!graphColoringUtil(graph, m, color, 0)) {
        cout << "Solution does not exist.";
        return;
    }

    char colorName[] = {' ', 'R', 'G', 'B'};

    cout << "Solution:\n";
    cout << "Vertex a -> " << colorName[color[0]] << endl;
    cout << "Vertex b -> " << colorName[color[1]] << endl;
    cout << "Vertex c -> " << colorName[color[2]] << endl;
    cout << "Vertex d -> " << colorName[color[3]] << endl;
}

int main() {

    // Adjacency Matrix for given graph
    vector<vector<int>> graph = {
        {0,1,1,1},  // a connected to b, c, d
        {1,0,1,0},  // b connected to a, c
        {1,1,0,1},  // c connected to a, b, d
        {1,0,1,0}   // d connected to a, c
    };

    int m = 3;  // Number of colors

    graphColoring(graph, m);

    return 0;
}
