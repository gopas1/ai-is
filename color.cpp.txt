#include <iostream>
#include <vector>
using namespace std;

// Check if assigning color c to vertex v is safe
bool isSafe(int v, vector<vector<int>>& graph,
            vector<int>& color, int c, int V) {
    
    for (int i = 0; i < V; i++) {
        if (graph[v][i] == 1 && color[i] == c)
            return false;
    }
    return true;
}

// Backtracking function
bool solveGraphColoring(vector<vector<int>>& graph,
                        int m, vector<int>& color,
                        int v, int V) {

    // If all vertices are colored
    if (v == V)
        return true;

    for (int c = 1; c <= m; c++) {
        if (isSafe(v, graph, color, c, V)) {
            color[v] = c;

            if (solveGraphColoring(graph, m, color, v + 1, V))
                return true;

            // Backtrack
            color[v] = 0;
        }
    }

    return false;
}

int main() {
    int V;
    cout << "Enter number of vertices: ";
    cin >> V;

    vector<vector<int>> graph(V, vector<int>(V));

    cout << "Enter adjacency matrix:\n";
    for (int i = 0; i < V; i++)
        for (int j = 0; j < V; j++)
            cin >> graph[i][j];

    int m;
    cout << "Enter number of colors: ";
    cin >> m;

    vector<int> color(V, 0);

    if (solveGraphColoring(graph, m, color, 0, V)) {
        cout << "\nSolution Exists:\n";
        for (int i = 0; i < V; i++)
            cout << "Vertex " << i
                 << " -> Color " << color[i] << endl;
    } else {
        cout << "\nNo solution exists with "
             << m << " colors.\n";
    }

    return 0;
}
